file(GLOB PARAGUI_SRC "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
find_package(PkgConfig QUIET)
if(PKG_CONFIG_FOUND)
  pkg_check_modules(SIGC sigc++-2.0 QUIET)
endif()
find_package(Freetype REQUIRED)
include_directories(${FREETYPE_INCLUDE_DIRS})

if(PARAGUI_SRC)
  if(NOT SIGC_FOUND)
    message(WARNING "sigc++ not found; skipping paragui build (install libsigc++-2.0-dev to enable)")
  else()
    add_library(paragui STATIC ${PARAGUI_SRC})
    target_include_directories(paragui PRIVATE ${CMAKE_SOURCE_DIR}/source ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_include_directories(paragui PRIVATE ${SIGC_INCLUDE_DIRS})
    target_link_libraries(paragui PRIVATE ${SIGC_LIBRARIES})
    if(DEFINED ASC_SDL_INCLUDE_DIR)
      target_include_directories(paragui PRIVATE ${ASC_SDL_INCLUDE_DIR})
    endif()
  endif()
endif()
